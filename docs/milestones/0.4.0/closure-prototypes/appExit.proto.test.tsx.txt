// @vitest-environment jsdom
import { afterEach, beforeAll, describe, expect, it, vi } from "vitest";
import { act, cleanup, render, screen } from "@testing-library/react";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";

vi.mock("react-i18next", async () => (await import("./host/mocks")).reactI18nextMock);
vi.mock("sonner", async () => (await import("./host/mocks")).sonnerMock);
vi.mock(
  "C:/Projects/cli-launchpad/src/components/PtyTerminal.tsx",
  async () => (await import("./host/mocks")).ptyTerminalMock,
);

import { tauriMock } from "C:/Projects/cli-launchpad/src/test/tauriMock";
import { App } from "C:/Projects/cli-launchpad/src/App";
import { useAppStore } from "C:/Projects/cli-launchpad/src/store/appStore";
import { registerBuiltinWorkspaceContentAdapters } from "C:/Projects/cli-launchpad/src/components/workspaceContentAdapters/builtins";
import { CapturingBoundary, createBackend, directory, flush, installBackend } from "./host/harness";

beforeAll(() => {
  window.matchMedia ??= ((query: string) => ({
    matches: false, media: query, onchange: null,
    addEventListener() {}, removeEventListener() {}, addListener() {}, removeListener() {},
    dispatchEvent: () => false,
  })) as unknown as typeof window.matchMedia;
  globalThis.ResizeObserver ??= class {
    observe() {}
    unobserve() {}
    disconnect() {}
  } as unknown as typeof ResizeObserver;
});
afterEach(() => cleanup());

describe("App exit flow prototype", () => {
  it("does not leave the dialog pending after a silent exit failure (S1A-N03)", async () => {
    const backend = createBackend();
    backend.handlers.set("detect_cli_status", () => []);
    backend.handlers.set("list_execution_tasks", () => []);
    backend.handlers.set("confirm_app_exit", () => {
      throw { code: "pty_sessions_active", message: "still running" };
    });
    installBackend(backend);
    const unregister = registerBuiltinWorkspaceContentAdapters();
    useAppStore.setState({ view: "projects", selectedDirectoryId: null });
    const errors: unknown[] = [];
    render(
      <QueryClientProvider client={new QueryClient({ defaultOptions: { queries: { retry: false } } })}>
        <CapturingBoundary errors={errors}>
          <App />
        </CapturingBoundary>
      </QueryClientProvider>,
    );
    await flush(10);
    console.log("errors after mount", errors.map(String));
    await act(async () => {
      tauriMock.emitEvent("app-exit-requested", { ptyCount: 0 }, "main");
    });
    await flush(10);
    const dialog = screen.queryByRole("dialog");
    console.log("dialog shown", Boolean(dialog));
    const buttons = dialog ? Array.from(dialog.querySelectorAll("button")).map((b) => [b.textContent, (b as HTMLButtonElement).disabled]) : [];
    console.log("buttons", JSON.stringify(buttons));
    unregister();
    expect(directory.id).toBe(1);
  });
});
