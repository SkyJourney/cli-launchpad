// @vitest-environment jsdom
import { afterEach, beforeAll, describe, expect, it, vi } from "vitest";
import { act, cleanup } from "@testing-library/react";

vi.mock("react-i18next", async () => (await import("./host/mocks")).reactI18nextMock);
vi.mock("sonner", async () => (await import("./host/mocks")).sonnerMock);
vi.mock(
  "C:/Projects/cli-launchpad/src/components/PtyTerminal.tsx",
  async () => (await import("./host/mocks")).ptyTerminalMock,
);

import { tauriMock } from "C:/Projects/cli-launchpad/src/test/tauriMock";
import { emitToMain, flush, mountWorkspace } from "./host/harness";

beforeAll(() => {
  globalThis.ResizeObserver ??= class {
    observe() {}
    unobserve() {}
    disconnect() {}
  } as unknown as typeof ResizeObserver;
});
afterEach(() => {
  vi.useRealTimers();
  cleanup();
});

describe("timer-driven host prototype", () => {
  it("bounds owner-lost reattach retries and stops after unmount (S1A-N05)", async () => {
    vi.spyOn(console, "warn").mockImplementation(() => undefined);
    const host = await mountWorkspace();
    await act(async () => host.ctx().launchSession(1, "claude"));
    await flush(10);
    const slot = host.ctx().slots[0];
    const terminal = host.ctx().terminalRefs.current.get(slot.instanceId) as any;
    terminal.reattachLostSession.mockRejectedValue(new Error("not yet"));
    host.backend.handlers.set("get_pty_session_window_status", () => "running");
    vi.useFakeTimers({ toFake: ["setTimeout", "clearTimeout", "setInterval", "clearInterval", "Date"] });
    await act(async () => {
      tauriMock.emitEvent("pty-session-owner-lost", { sessionId: slot.sessionId }, "main");
    });
    await act(async () => {
      await vi.advanceTimersByTimeAsync(30_000);
    });
    const during = terminal.reattachLostSession.mock.calls.length;
    console.log("timer count before unmount", vi.getTimerCount());
    host.dispose();
    await act(async () => {
      await vi.advanceTimersByTimeAsync(30_000);
    });
    const after = terminal.reattachLostSession.mock.calls.length;
    console.log("timer count after unmount", vi.getTimerCount());
    console.log("reattach attempts within 30s", during, "after unmount +30s", after - during);
  });

  it("treats a detached file whose child never acknowledges flush as dirty", async () => {
    const host = await mountWorkspace();
    await act(async () => {
      await host.ctx().openProjectFile(1, "C:/project", "a.txt");
    });
    await flush();
    const document = host.ctx().fileDocuments[0];
    await act(async () => {
      void host.ctx().detachFile(document.id).catch(() => undefined);
    });
    await flush();
    const created = tauriMock.state.createdWindows.at(-1)!;
    const token = new URL((created.options as any).url, "http://x").searchParams.get("fileHandoffToken")!;
    await emitToMain("workspace-file-window-ready", { documentId: document.id, token, windowLabel: created.label });
    await flush();
    await emitToMain("workspace-file-window-attached", { documentId: document.id, token, windowLabel: created.label });
    await flush();
    vi.useFakeTimers({ toFake: ["setTimeout", "clearTimeout"] });
    let impacts: any;
    await act(async () => {
      const pending = host.ctx().collectExitImpacts(0).then((value) => (impacts = value));
      await vi.advanceTimersByTimeAsync(600);
      await pending;
    });
    console.log("impacts", JSON.stringify(impacts));
    expect(impacts.dirtyFiles.map((f: any) => f.documentId)).toEqual([document.id]);
    host.dispose();
  });
});
