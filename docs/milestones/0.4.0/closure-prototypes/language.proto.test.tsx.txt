// @vitest-environment jsdom
import { afterEach, beforeAll, describe, expect, it, vi } from "vitest";
import { act, cleanup } from "@testing-library/react";

vi.mock("sonner", async () => (await import("./host/mocks")).sonnerMock);
vi.mock(
  "C:/Projects/cli-launchpad/src/components/PtyTerminal.tsx",
  async () => (await import("./host/mocks")).ptyTerminalMock,
);

import { i18n } from "C:/Projects/cli-launchpad/src/i18n";
import { tauriMock } from "C:/Projects/cli-launchpad/src/test/tauriMock";
import { flush, mountWorkspace } from "./host/harness";

beforeAll(() => {
  globalThis.ResizeObserver ??= class {
    observe() {}
    unobserve() {}
    disconnect() {}
  } as unknown as typeof ResizeObserver;
});
afterEach(() => cleanup());

describe("language change prototype", () => {
  it("keeps open buffers and running PTYs when the UI language changes", async () => {
    await i18n.changeLanguage("en");
    const host = await mountWorkspace();
    await act(async () => {
      await host.ctx().openProjectFile(1, "C:/project", "a.txt");
    });
    await flush();
    await act(async () => {
      host.ctx().launchSession(1, "claude");
    });
    await flush(10);
    const document = host.ctx().fileDocuments[0];
    await act(async () => {
      host.ctx().editFile(document.id, "unsaved edit");
    });
    await flush();
    const slot = host.ctx().slots[0];
    host.backend.layout = {
      ...host.backend.layout,
      status: { status: "ready" },
      revision: host.backend.saved.at(-1)?.revision ?? 1,
      layout: host.backend.saved.at(-1)?.layout ?? null,
      slotStates: [
        { instanceId: slot.instanceId, state: "running", currentProjectName: "Project" },
      ],
    };
    const readsBefore = tauriMock.state.invokeCalls.filter((c) => c.command === "get_workspace_layout").length;
    console.log("before: buffer", host.ctx().fileBuffers[document.id]?.content, "slots", host.ctx().slots.length);
    await act(async () => {
      await i18n.changeLanguage("zh");
    });
    await flush(10);
    const readsAfter = tauriMock.state.invokeCalls.filter((c) => c.command === "get_workspace_layout").length;
    console.log("layout reads before/after", readsBefore, readsAfter);
    console.log("after: buffer", host.ctx().fileBuffers[document.id]?.content, "docs", host.ctx().fileDocuments.length, "slots", host.ctx().slots.length);
    host.dispose();
    expect(true).toBe(true);
  });
});
